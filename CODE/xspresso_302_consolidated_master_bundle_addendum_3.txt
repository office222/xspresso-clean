const DEFAULT_EMAIL_PREFERENCES = Object.freeze({
  campaignUpdates: false,
  monitoringAlerts: false,
  leadFollowups: false,
  creativeAssets: false,
});
const EMAIL_PREFERENCE_KEYS = new Set(Object.keys(DEFAULT_EMAIL_PREFERENCES));

function fail(status, code, message) {
  return Object.assign(new Error(message), { status, code });
}
function scopeFor(user) {
  if (!user?.id || !user?.tenantId) throw fail(401, 'UNAUTHENTICATED', 'Authentication required.');
  return { userId: user.id, tenantId: user.tenantId };
}
function eventTime(event) {
  const value = event?.createdAt || event?.at || event?.updatedAt;
  const ms = Date.parse(value || '');
  return Number.isFinite(ms) ? ms : 0;
}
function eventCategory(event) {
  const text = `${event?.kind || ''} ${event?.action || ''} ${event?.type || ''}`.toLowerCase();
  if (text.includes('alert') || text.includes('monitor')) return 'monitoringAlerts';
  if (text.includes('followup') || text.includes('lead')) return 'leadFollowups';
  if (text.includes('asset') || text.includes('image') || text.includes('creative')) return 'creativeAssets';
  return 'campaignUpdates';
}
function safeTitle(event) {
  const value = String(event?.action || event?.kind || 'account activity')
    .replace(/[^a-z0-9._ -]/gi, ' ')
    .replace(/[._-]+/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
  return value.slice(0, 100) || 'Account activity';
}
function compareCursor(event, cursor) {
  const time = eventTime(event);
  if (time !== cursor.lastEventAt) return time > cursor.lastEventAt;
  return String(event.id || '').localeCompare(String(cursor.lastEventId || '')) > 0;
}

export class NotificationEmailService {
  constructor({ store, mailer = null, clock = () => Date.now(), dashboardUrl = '/' }) {
    if (!store?.read || !store?.update) throw new TypeError('store must support read and update');
    this.store = store;
    this.mailer = mailer;
    this.clock = clock;
    this.dashboardUrl = dashboardUrl;
  }

  async getPreferences(user) {
    const scope = scopeFor(user);
    const state = await this.store.read();
    const row = (state.notificationEmailPreferences || []).find(
      item => item.userId === scope.userId && item.tenantId === scope.tenantId,
    );
    return {
      preferences: { ...DEFAULT_EMAIL_PREFERENCES, ...(row?.preferences || {}) },
      emailAvailable: Boolean(this.mailer),
      enabledAt: row?.enabledAt || null,
      updatedAt: row?.updatedAt || null,
    };
  }

  async updatePreferences(user, patch) {
    const scope = scopeFor(user);
    if (!patch || typeof patch !== 'object' || Array.isArray(patch) || !Object.keys(patch).length) {
      throw fail(400, 'INVALID_PREFERENCES', 'Provide at least one email preference.');
    }
    for (const [key, value] of Object.entries(patch)) {
      if (!EMAIL_PREFERENCE_KEYS.has(key) || typeof value !== 'boolean') {
        throw fail(400, 'INVALID_PREFERENCES', `Unsupported email preference: ${key}.`);
      }
    }
    const now = new Date(this.clock()).toISOString();
    return this.store.update(state => {
      state.notificationEmailPreferences ||= [];
      let row = state.notificationEmailPreferences.find(
        item => item.userId === scope.userId && item.tenantId === scope.tenantId,
      );
      if (!row) {
        row = { ...scope, preferences: { ...DEFAULT_EMAIL_PREFERENCES }, createdAt: now, enabledAt: null };
        state.notificationEmailPreferences.push(row);
      }
      const previouslyEnabled = Object.values(row.preferences || {}).some(Boolean);
      row.preferences = { ...DEFAULT_EMAIL_PREFERENCES, ...(row.preferences || {}), ...patch };
      const isEnabled = Object.values(row.preferences).some(Boolean);
      if (!previouslyEnabled && isEnabled) row.enabledAt = now;
      row.updatedAt = now;
      return {
        preferences: { ...row.preferences },
        emailAvailable: Boolean(this.mailer),
        enabledAt: row.enabledAt,
        updatedAt: row.updatedAt,
      };
    });
  }

  async runOne() {
    if (!this.mailer) return { sent: false, reason: 'email_not_configured' };
    const snapshot = await this.store.read();
    const users = snapshot.users || [];
    const preferenceRows = snapshot.notificationEmailPreferences || [];
    const cursors = snapshot.notificationEmailCursors || [];
    const audit = snapshot.auditEvents || [];
    const alerts = snapshot.alerts || [];

    const events = [
      ...audit.map((event, index) => ({ ...event, id: String(event.id || `audit-${index}`) })),
      ...alerts.map(event => ({
        id: `alert-${event.id}-${event.status || 'open'}`,
        tenantId: event.tenantId,
        action: event.status === 'resolved' ? 'monitoring.alert_resolved' : 'monitoring.alert_opened',
        createdAt: event.updatedAt || event.createdAt,
        kind: 'alert',
      })),
    ];

    for (const user of users) {
      if (!user?.id || !user?.tenantId || !user?.email) continue;
      const prefRow = preferenceRows.find(row => row.userId === user.id && row.tenantId === user.tenantId);
      const preferences = { ...DEFAULT_EMAIL_PREFERENCES, ...(prefRow?.preferences || {}) };
      if (!Object.values(preferences).some(Boolean)) continue;
      const cursor = cursors.find(row => row.userId === user.id && row.tenantId === user.tenantId) || {
        lastEventAt: Date.parse(prefRow?.enabledAt || prefRow?.createdAt || new Date(this.clock()).toISOString()),
        lastEventId: '',
      };
      const eligible = events
        .filter(event => event.tenantId === user.tenantId && preferences[eventCategory(event)] === true && compareCursor(event, cursor))
        .sort((a, b) => eventTime(a) - eventTime(b) || String(a.id).localeCompare(String(b.id)))
        .slice(0, 20);
      if (!eligible.length) continue;

      const last = eligible[eligible.length - 1];
      const digestKey = `${user.tenantId}:${user.id}:${eligible.map(item => item.id).join(',')}`;
      const safeEvents = eligible.map(event => `• ${safeTitle(event)} — ${new Date(eventTime(event)).toISOString()}`);
      const text = [
        'Xspresso account activity',
        '',
        ...safeEvents,
        '',
        `Open your Xspresso dashboard: ${this.dashboardUrl}`,
        'You can change email notification preferences in the dashboard.',
      ].join('\n');

      await this.mailer.send({
        to: user.email,
        subject: `Xspresso activity update (${eligible.length})`,
        text,
        idempotencyKey: `activity-digest-${digestKey}`.slice(0, 160),
      });

      const sentAt = new Date(this.clock()).toISOString();
      await this.store.update(state => {
        state.notificationEmailCursors ||= [];
        let row = state.notificationEmailCursors.find(item => item.userId === user.id && item.tenantId === user.tenantId);
        if (!row) {
          row = { userId: user.id, tenantId: user.tenantId };
          state.notificationEmailCursors.push(row);
        }
        row.lastEventAt = eventTime(last);
        row.lastEventId = String(last.id);
        row.updatedAt = sentAt;
        state.notificationEmailDeliveries ||= [];
        state.notificationEmailDeliveries.push({
          id: digestKey,
          userId: user.id,
          tenantId: user.tenantId,
          eventCount: eligible.length,
          sentAt,
        });
        if (state.notificationEmailDeliveries.length > 5000) {
          state.notificationEmailDeliveries.splice(0, state.notificationEmailDeliveries.length - 5000);
        }
        return true;
      });
      return { sent: true, userId: user.id, eventCount: eligible.length };
    }
    return { sent: false, reason: 'nothing_due' };
  }
}

export class NotificationEmailScheduler {
  constructor({ worker, intervalMs = 60_000, batchLimit = 20, onError = () => {}, timers = globalThis }) {
    if (!worker?.runOne) throw new TypeError('worker.runOne is required');
    this.worker = worker;
    this.intervalMs = intervalMs;
    this.batchLimit = batchLimit;
    this.onError = onError;
    this.timers = timers;
    this.timer = null;
    this.active = null;
  }
  async tick() {
    if (this.active) return { skipped: true };
    this.active = (async () => {
      const results = [];
      for (let i = 0; i < this.batchLimit; i += 1) {
        const result = await this.worker.runOne();
        results.push(result);
        if (!result?.sent) break;
      }
      return results;
    })();
    try { return await this.active; }
    finally { this.active = null; }
  }
  start() {
    if (this.timer) return false;
    void this.tick().catch(this.onError);
    this.timer = this.timers.setInterval(() => { void this.tick().catch(this.onError); }, this.intervalMs);
    this.timer?.unref?.();
    return true;
  }
  async stop() {
    if (this.timer) this.timers.clearInterval(this.timer);
    this.timer = null;
    if (this.active) await this.active.catch(this.onError);
  }
}

function send(res, status, payload) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store', 'X-Content-Type-Options': 'nosniff' });
  res.end(JSON.stringify(payload));
}
function bearer(req) {
  const match = /^Bearer ([A-Za-z0-9_-]{20,512})$/.exec(req.headers?.authorization || '');
  return match?.[1] || null;
}
async function readJson(req, maxBytes = 16 * 1024) {
  const chunks = []; let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > maxBytes) throw Object.assign(new Error('Request too large.'), { status: 413, code: 'BODY_TOO_LARGE' });
    chunks.push(chunk);
  }
  if (!size) return {};
  try { return JSON.parse(Buffer.concat(chunks).toString('utf8')); }
  catch { throw Object.assign(new Error('Invalid JSON.'), { status: 400, code: 'INVALID_JSON' }); }
}

export function createNotificationEmailApiHandler({ auth, service, fallback }) {
  if (!auth?.authenticate || !service) throw new TypeError('auth and service are required');
  return async function notificationEmailApi(req, res) {
    const url = new URL(req.url || '/', 'http://localhost');
    if (url.pathname !== '/api/notifications/email-preferences') {
      return fallback ? fallback(req, res) : send(res, 404, { error: 'Not found.' });
    }
    try {
      const token = bearer(req);
      if (!token) return send(res, 401, { error: 'Authentication required.' });
      const user = await auth.authenticate(token);
      if (!user) return send(res, 401, { error: 'Authentication required.' });
      if (req.method === 'GET') return send(res, 200, await service.getPreferences(user));
      if (req.method === 'PATCH') {
        const body = await readJson(req);
        return send(res, 200, await service.updatePreferences(user, body.preferences));
      }
      return send(res, 405, { error: 'Method not allowed.' });
    } catch (error) {
      const status = Number.isInteger(error.status) ? error.status : 500;
      return send(res, status, { error: status >= 500 ? 'Email preferences request failed.' : error.message, code: error.code || 'EMAIL_PREFERENCES_ERROR' });
    }
  };
}

async function callApi(path, options = {}) {
  const token = sessionStorage.getItem(TOKEN_KEY);
  if (!token) throw new Error('Sign in to manage email notifications.');
  const response = await fetch(path, {
    ...options,
    headers: { Authorization: `Bearer ${token}`, ...(options.body ? { 'Content-Type': 'application/json' } : {}), ...(options.headers || {}) },
  });
  let result = {};
  try { result = await response.json(); } catch {}

  });

  const server = createServer(async (req, res) => {
    const url = new URL(req.url || '/', 'http://localhost');

      const decision = limiter.consume(req.socket?.remoteAddress || 'unknown');
      if (!decision.allowed) {
        res.setHeader('Retry-After', String(decision.retryAfterSeconds));
        return send(res, 429, JSON.stringify({ error: 'Too many requests.' }), { 'Content-Type': 'application/json; charset=utf-8' });
      }
    }
    if (url.pathname.startsWith('/api/')) {
      try { return await api(req, res); }
      catch { if (!res.headersSent) return send(res, 500, JSON.stringify({ error: 'Internal server error.' }), { 'Content-Type': 'application/json; charset=utf-8' }); return; }

    const item = ASSETS.get(url.pathname);
    if (!item) return send(res, 404, 'Not found.');
    try {
      let body = await readFile(join(webRoot, item[0]));
      if (url.pathname === '/' || url.pathname.endsWith('.html')) {
        let html = body.toString('utf8');

        const email = '<script type="module" src="/xspresso_notification_email_panel_v1.js"></script>';

        if (!html.includes(email)) html = html.replace('</body>', `${email}</body>`);
        body = Buffer.from(html);
      }
      return send(res, 200, req.method === 'HEAD' ? '' : body, {
        'Content-Type': item[1],
        'Cache-Control': 'no-store',
        'Content-Security-Policy': "default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self' blob: data:; connect-src 'self'; base-uri 'none'; frame-ancestors 'none'",
      });
    } catch { return send(res, 500, 'Unable to load application asset.'); }
  });
  const limiter = new AuthRateLimiter({ maxAttempts: 10, windowMs: 60_000 });
  const config = options.config || base.config;
  let listening = false;
  return {
    ...base,
    api,
    server,
    notificationEmailService: emailService,
    emailScheduler: scheduler,
    emailAvailable: Boolean(mailer),

  };
}
